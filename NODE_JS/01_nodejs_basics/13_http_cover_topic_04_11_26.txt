const http = require("http");

const port = 3000;

const App = http.createServer((req, res) => {

    if (req.method === "GET" && req.url === "/") {
        res.end("home")
    }
    else if (req.method === "GET"&&req.url === "/about"){
        res.end("about page")
    }
   
     else {
        res.end("page not found .")
    }

    
});

App.listen(port, () => {
    console.log(`http://localhost:${port}`);
})
